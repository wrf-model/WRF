cmake_minimum_required(VERSION 3.20)
project(WRF_RRTMGP_COLUMN_TEST LANGUAGES Fortran)
enable_testing()
get_filename_component(WRF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${WRF_ROOT}/external/rte_rrtmgp" rte_rrtmgp)
set(adapter "${WRF_ROOT}/phys/module_ra_rrtmgp.F")
set_source_files_properties("${adapter}" PROPERTIES Fortran_FORMAT FREE Fortran_PREPROCESS ON)
add_executable(test_rrtmgp_columns standalone_wrf_error.f90 "${adapter}" test_columns.f90)
target_link_libraries(test_rrtmgp_columns PRIVATE wrf_rrtmgp)
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  target_compile_options(test_rrtmgp_columns PRIVATE -ffree-line-length-none)
endif()
set(RRTMGP_DATA_DIR "${WRF_ROOT}/run" CACHE PATH "RRTMGP coefficient directory")
add_test(NAME wrf_rrtmgp_columns COMMAND test_rrtmgp_columns "${RRTMGP_DATA_DIR}")
